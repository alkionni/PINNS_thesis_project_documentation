class DataModule(pl.LightningDataModule):
    """
    Data module for handling the dataset.
    """
    def __init__(self, path, label, oria, dataset_size, batch_size=1024,
                 num_workers=4, train_perc=0.8, valid_perc=0.1,
                 train_perc_to_keep=1
                 ):
        """
        Initialize the data module.
        :param path: The path to the saved dataset.
        :param label: Which label to use (LDR or LNR).
        :param oria: The input ranges.
        :param dataset_size: The dataset size.
        :param batch_size: The batch size.

        :param train_perc: The percentage of the dataset to keep for training.
        :param valid_perc: The percentage of the dataset to keep for validation.

        self.path = path
        self.label = label
        self.batch_size = batch_size
        self.num_workers = num_workers
        self.dataset_size = dataset_size
        self.train_perc = train_perc
        self.val_perc = valid_perc
        self.train_sbt = []
        self.val_sbt = []
        self.test_sbt = []
        self.label_range = []
        self.label_mean = []
        self.oria = oria
        self.train_perc_to_keep = train_perc_to_keep
        self.train_sbt_size_to_keep = None

    def setup(self, stage) -> None:
        """
        Prepare dataset for training. Load the raw dataset and split it into train,
        validation and test sets.
        :param stage: Needed by default.
        :return: None
        """

        # Create raw dataset instance.
        entire_dataset = RawDataset(self.path, self.dataset_size, self.oria, self.label)
        dataset_len = len(entire_dataset)

        # Create the subset sizes based on the given percentages.
        train_size = int(self.train_perc * dataset_len)
        valid_size = int(self.val_perc * dataset_len)
        test_size = dataset_len - train_size - valid_size

        # Split dataset into train, validation and test subsets.
        self.train_sbt, self.val_sbt, self.test_sbt = random_split(
            entire_dataset,
            [train_size, valid_size, test_size],
            generator=torch.Generator().manual_seed(0)
        )

        # Keep the percentage of the train subset for sample efficiency testing (if
        # the train precentage to keep is not one).
        self.train_sbt_size_to_keep = int(self.train_perc_to_keep * train_size)
        self.train_sbt.indices = self.train_sbt.indices[:self.train_sbt_size_to_keep]  # Keep the first samples

        # Get the label's range and mean.
        self.label_range = entire_dataset.target_range
        self.label_mean = entire_dataset.target_mean

        print(self.label_range)
        print(f"The train subset size: {self.train_sbt_size_to_keep}")

class RawDataset(Dataset):
    """
    Class for loading the raw dataset.
    """
    def __init__(self, path, dataset_size, oria, label):
        """
        Initialize the dataset.
        :param path: The path to the saved dataset.
        :param dataset_size: The dataset size.
        :param oria: The input range.
        :param label: The label to use (LDR or LNR).
        """
        self.path = path
        self.oria = oria

        # Load dataset from file.
        with h5py.File(self.path, "r") as f:
            inputs = f['input'][:dataset_size]
            ldr = f['ldr'][:dataset_size]
            lnr = f['lnr'][:dataset_size]

            # Create a mask for filtering bad samples. Double mask: 76413 samples.
            mask1 = np.abs(ldr[:, 0] - 1.8) <= 0.2
            mask2 = np.abs(lnr[:, 0] - 1.8) <= 0.2
            mask = mask1 & mask2

            # Choose which label to use.
            if label == "LDR":
                self.target = ldr[mask]
            elif label == "LNR":
                self.target = lnr[mask]
            else:
                self.target = None
                raise RuntimeError("Label was not defined. "
                                   "Please choose between 'LDR' and 'LNR'"
                                   "in the config file.")
            self.inputs = inputs[mask]

        # Notes for other sample filters.
        # Excluding only zero-voltage ldr.
        # mask = np.all(np.isclose(self.ldr, 0, atol=1e-3), axis=1)
        # self.inputs = self.inputs[~mask]
        # self.ldr = self.ldr[~mask]

        # Excluding all non-func ldr: 85272.
        # mask = np.abs(self.ldr[:, 0] - 1.8) <= 0.2
        # self.inputs = self.inputs[mask]
        # self.ldr = self.ldr[mask]

        # Store dataset length, target range and mean.
        self.dataset_len = len(self.inputs)
        self.target_range = self.target.min(), self.target.max()
        self.target_mean = np.mean(self.target, axis=0)
        print(self.target_range)

    def __len__(self) -> int:
        """
        Get dataset length.
        :return: The number of samples the dataset has.
        """
        return self.dataset_len

    def __getitem__(self, idx) -> tuple:
        """
        Get a single transformed dataset sample.
        :param idx: The index of the sample.
        :return: The feature and label of the sample.
        """
        # Normalize sample.
        # Normalize feature.
        feature = np.zeros(self.inputs[idx].shape)
        for i, param in enumerate(self.oria):
            feature[i] = \
                (2 * (self.inputs[idx][i] - self.oria[param][0]) /
                 (self.oria[param][1] -
                  self.oria[param][0]) - 1)
        # Normalize label.
        label = 2 * (self.target[idx] - self.target_range[0]) / (self.target_range[1] - self.target_range[0]) - 1

        # Transform both into tensors.
        feature = torch.tensor(feature, dtype=torch.float32)
        label = torch.tensor(label, dtype=torch.float32)

        return feature, label
