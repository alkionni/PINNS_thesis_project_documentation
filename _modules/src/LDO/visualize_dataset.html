def normalize_inputs(dataset):
    """Normalize the entire dataset's features."""
    feature = np.zeros(dataset.inputs.shape)
    for i, param in enumerate(dataset.oria):
        feature[:, i] = \
            (2 * (dataset.inputs[:, i] - dataset.oria[param][0]) /
             (dataset.oria[param][1] -
              dataset.oria[param][0]) - 1)
    return feature

def normalize_labels(dataset):
    """Normalize the entire dataset's labels."""
    label = (2 * (dataset.target - dataset.target_range[0]) /
             (dataset.target_range[1] - dataset.target_range[0]) - 1)
    return label

def plot_predictions_distribution(predictions, real_pred):
    """Plot the before and after de-normalization distribution of the predictions."""
    if config.LABEL == 'LDR':
        label = "LDR"
    elif config.LABEL == 'LNR':
        label = "LNR"
    else:
        label = None
        raise RuntimeError("Label was not defined. "
                           "Please choose between 'LDR' and 'LNR'"
                           "in the config file.")

    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
    fig.suptitle(f'Distribution of {label} prediction before and after de-normalization', fontsize=14)
    bins = np.linspace(predictions.min(), predictions.max(), 100)
    ax1.set_title(f"Before")
    ax1.set_xlabel(f"{label} (per unit)")
    ax1.hist(predictions.flatten(), bins, edgecolor='black', alpha=0.7)
    ax1.set_xscale('linear')

    bins = np.linspace(real_pred.min(), real_pred.max(), 100)
    ax2.set_title(f"After")
    ax2.set_xlabel(f"{label} (Volt)")
    ax2.hist(real_pred.flatten(), bins, edgecolor='black', alpha=0.7)
    ax2.set_xscale('linear')
    plt.show()

def visualize_dataset():
    # Initialize data module.
    dt = RawDataset(
        path=config.DATASET_PATH,
        dataset_size=config.DATASET_SIZE,
        oria=config.ORIA,
        label=config.LABEL
    )

    # Normalize the dataset.
    features = normalize_inputs(dt)
    labels = normalize_labels(dt)

    # ------------ PLOT FEATURES DISTRIBUTION -------------------------
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
    fig.suptitle(f'Distribution of the 18 design parameters before and after normalization', fontsize=14)
    bins = np.logspace(-12, 6, 100)
    ax1.set_title(f"Before")
    ax1.set_xlabel("Widths (m) & Capacitances (F)")
    ax1.hist(dt.inputs.flatten(), bins, edgecolor='black', alpha=0.7)
    ax1.set_xscale('log')

    bins = np.linspace(features.min(), features.max(), 100)
    ax2.set_title(f"After")
    ax2.set_xlabel("Widths (per unit) & Capacitances (per unit)")
    ax2.hist(features.flatten(), bins, edgecolor='black', alpha=0.7)
    ax2.set_xscale('linear')
    plt.show()

    # ------------ PLOT LABEL DISTRIBUTION -------------------------
    if config.LABEL == 'LDR':
        label = "LDR"
    elif config.LABEL == 'LNR':
        label = "LNR"
    else:
        label = None
        raise RuntimeError("Label was not defined. "
                           "Please choose between 'LDR' and 'LNR'"
                           "in the config file.")
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
    fig.suptitle(f'Distribution of {label} before and after normalization', fontsize=14)
    bins = np.linspace(dt.target_range[0], dt.target_range[1], 100)
    ax1.set_title(f"Before")
    ax1.set_xlabel(f"{label} (Volt)")
    ax1.hist(dt.target.flatten(), bins, edgecolor='black', alpha=0.7)
    ax1.set_xscale('linear')

    bins = np.linspace(labels.min(), labels.max(), 100)
    ax2.set_title(f"After")
    ax2.set_xlabel(f"{label} (per unit)")
    ax2.hist(labels.flatten(), bins, edgecolor='black', alpha=0.7)
    ax2.set_xscale('linear')
    plt.show()

    print("The end.")




if __name__ == '__main__':
    visualize_dataset()
