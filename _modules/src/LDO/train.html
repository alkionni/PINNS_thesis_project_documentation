def train(min_epochs: int, max_epochs: int, logger: bool) -> None:
    """Train the neural network."""
    if logger:
        logger = WandbLogger(project="project-name", name="model-v0")

    # Initialize data module.
    dm = DataModule(
        path=config.DATASET_PATH,
        label=config.LABEL,
        dataset_size=config.DATASET_SIZE,
        oria=config.ORIA,
        batch_size=512
    )

    # Initialize early stopping callback.
    early_stop_callback = EarlyStopping(
        monitor="val_loss",
        min_delta=0.00,
        patience=1000,
        verbose=False,
        mode="min"
    )

    # Initialize model checkpoint callback.
    checkpoint_callback = ModelCheckpoint(
        monitor='val_loss',
        mode='min',
        # save_top_k=10,
        # save_last=True,
        filename="best-model-{epoch:02d}-{val_loss:.4f}"
    )

    # Initialize model.
    model = ResidualMLP(
        lr=0.002,
        weight_decay=0.0004,
        signal_lambda=3,
        hidden_dim=256,
        num_blocks=5,
    )

    # Initialize trainer
    trainer = pl.Trainer(
        # overfit_batches=1,
        # fast_dev_run=False,
        # limit_train_batches=10,
        # limit_val_batches=0,
        # num_sanity_val_steps=0,
        # logger=False,
        logger=logger,
        accelerator='gpu',
        devices=1,
        min_epochs=min_epochs,
        max_epochs=max_epochs,
        precision=32,
        callbacks=[
            MyPrintingCallback(),
            DisableValProgressBar(),
            early_stop_callback,
            checkpoint_callback
        ],
        # gradient_clip_val=0.5,
        # gradient_clip_algorithm="norm"
    )

    # Train, validate and test.
    trainer.fit(model, dm)
    trainer.validate(model, dm)
    trainer.test(model, dm)

    print("End of training.")




if __name__ == '__main__':
    train(config.MIN_EPOCHS, config.MAX_EPOCHS, True)
