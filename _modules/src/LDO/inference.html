def inference(num_plots: int):
    """Inference on test set."""
    # Initialize model.
    model = ResidualMLP.load_from_checkpoint(
        '/home/isidsouk/PINNS/src/LDO/ldo-ml-final/res-mlp-LDR-v0/checkpoints/best-model-epoch=5113-val_loss=0.0002.ckpt'
    )

    # Initialize data module.
    dm = DataModule(
        path=config.DATASET_PATH,
        dataset_size=config.DATASET_SIZE,
        oria=config.ORIA,
        label=config.LABEL
    )

    # Initialize trainer.
    trainer = pl.Trainer(
        accelerator='gpu',
        devices=1,
        min_epochs=1,
        max_epochs=10,
        precision=32,
        callbacks=[MyPrintingCallback()]
    )

    # Make predictions on test subset.
    predictions = trainer.predict(model, dm)

    # Extract and concatenate all predictions and labels.
    all_preds_bf = torch.cat([batch["preds"] for batch in predictions]).detach().cpu().numpy()
    all_labels_bf = torch.cat([batch["labels"] for batch in predictions]).detach().cpu().numpy()

    # Inverse transform.
    all_preds = (all_preds_bf + 1) / 2 * (config.TARGET[1] - config.TARGET[0]) + config.TARGET[0]
    all_labels = (all_labels_bf + 1) / 2 * (config.TARGET[1] - config.TARGET[0]) + config.TARGET[0]

    # Plot distribution before and after inverse transformation.
    plot_predictions_distribution(all_preds_bf, all_preds)

    # Get total prediction error indices.
    t_mse = np.mean((all_preds - all_labels) ** 2)
    t_rmse = np.sqrt(np.mean((all_preds - all_labels) ** 2))
    t_mae = mean_absolute_error(all_labels, all_preds)
    t_mape = mean_absolute_percentage_error(all_labels, all_preds)

    # Show results.
    tot_plots = len(predictions)
    time = np.linspace(0, 0.00016, 1000)
    if num_plots > tot_plots:
        num_plots = tot_plots
    for i in range(num_plots):
        mse = np.mean((all_preds[i] - all_labels[i]) ** 2)
        rmse = np.sqrt(np.mean((all_preds[i] - all_labels[i]) ** 2))
        mae = mean_absolute_error(all_labels[i], all_preds[i])
        mape = mean_absolute_percentage_error(all_labels[i], all_preds[i])

        plt.figure(figsize=(12, 5))
        plt.suptitle(f'Total mse: {t_mse}, mae: {t_mae}, mape: {t_mape}, rmse: {t_rmse}', fontsize=12)
        plt.title(f'Plot {i + 1}/{tot_plots}, mse: {mse}, mae: {mae}, mape: {mape}, rmse: {rmse}', fontsize=10)
        plt.plot(time, all_labels[i], color='blue', label='True')
        plt.plot(time, all_preds[i], color='red', label='Predicted')
        plt.plot(time, dm.label_mean, color='green', linestyle='--', label='Dataset mean')
        plt.xlabel('Time (s)')
        plt.ylabel('Vout (Volt)')
        plt.legend()
        plt.grid(True, which='both', ls='-', alpha=0.5)

        plt.show()




if __name__ == '__main__':
    inference(1)
