class MLP(pl.LightningModule):
    """
    The MLP neural network module.
    """
    def __init__(self, lr=1e-5, weight_decay=1e-4, signal_lambda=10):
        """
        Initialize the neural network.
        :param lr: The optimizer learning rate.
        :param weight_decay: The optimizer weight decay.
        :param signal_lambda: The loss function lambda parameter.
        """
        super().__init__()
        self.save_hyperparameters()

        # The pyramid MLP.
        self.network = nn.Sequential(
            nn.Linear(18, 64),
            nn.ReLU(),
            nn.Linear(64, 128),
            nn.ReLU(),
            nn.Linear(128, 256),
            nn.ReLU(),
            nn.Linear(256, 512),
            nn.ReLU(),
            nn.Linear(512, 1000),
        )

        # The 4 layer MLP.
        # self.network = nn.Sequential(
        #     nn.Linear(18, 1000),
        #     nn.ReLU(),
        #     nn.Linear(1000, 1000),
        #     nn.ReLU(),
        #     nn.Linear(1000, 1000),
        #     nn.ReLU(),
        #     nn.Linear(1000, 1000),
        # )
        self.mse = nn.MSELoss()

    def forward(self, x):
        """
        Forward pass: input shape (Batch, 18) -> output shape (Batch, 10000).
        """
        out = self.network(x)
        return out

    def training_step(self, batch, batch_idx):
        """
        What to do in every training step.
        :param batch: The current batch.
        :param batch_idx: The index of the current batch.
        :return: The train loss.
        """
        x, y = batch
        out = self.forward(x)

        # Calculate weighted mse loss.
        diff = y[:, 1:] - y[:, :-1]
        padded_diff = f.pad(diff, (1, 0), value=0)
        weights = 1 + self.hparams.signal_lambda * torch.abs(padded_diff)
        loss = torch.sum(weights * self.mse(y, out)) / torch.sum(weights)

        # Log only every epoch.
        self.log('train_loss', loss, on_step=False, on_epoch=True, prog_bar=True, logger=True)
        return loss

class UpBlock(nn.Module):
    """
    The 1D CNN block for up-sampling. Doubles the sequence length,
    then does a 1d convolution layer + BatchNorm + activation refine step.
    """

    def __init__(self, in_ch: int, out_ch: int, kernel_size: int = 7):
        """
        Initialize the block.
        :param in_ch: The input size.
        :param out_ch: The output size.
        :param kernel_size: The kernel size.
        """
        super().__init__()
        padding = kernel_size // 2
        self.block = nn.Sequential(
            nn.Upsample(scale_factor=2, mode="linear", align_corners=False),
            nn.Conv1d(in_ch, out_ch, kernel_size=kernel_size, padding=padding),
            nn.BatchNorm1d(out_ch),
            nn.LeakyReLU(0.2, inplace=True),
        )



[docs]
    def forward(self, x):
        """A forward pass through the block."""
        return self.block(x)

class CNNRegressor(pl.LightningModule):
    """
    The 1D CNN neural network module.
    """
    def __init__(
            self,
            input_dim: int = 18,
            seq_len: int = 1000,
            base_channels: int = 128,
            init_len: int = 8,
            lr=1e-5,
            weight_decay=1e-4,
            signal_lambda=10
    ):
        """
        Initialize the module.
        :param input_dim: The number of inputs.
        :param seq_len: The output sequence length.
        :param base_channels: The width of the widest (first) conv stage; controls
        overall model capacity/parameter count
        :param init_len: The length of the initial learned "seed"
        sequence before up-sampling begins
        :param lr: The optimizer learning rate.
        :param weight_decay: The optimizer weight decay.
        :param signal_lambda: The loss function lambda parameter.
        """
        super().__init__()
        self.save_hyperparameters()

        self.init_len = init_len
        self.init_channels = base_channels

        # Project the 18 parameters into a short, multichannel "seed" tile.
        self.fc = nn.Linear(input_dim, base_channels * init_len)

        # Channel schedule as we up-sample: 128 -> 64 -> 64 -> 32 -> 32 -> 16 -> 16 -> 8 -> 8
        c = base_channels
        channels = [c, c // 2, c // 2, c // 4, c // 4, c // 8, c // 8, c // 16]

        # Up-sample
        blocks = []
        for i in range(len(channels) - 1):
            blocks.append(UpBlock(channels[i], channels[i + 1]))
        self.up_blocks = nn.Sequential(*blocks)

        # After 6 UpBlocks: init_len * 2^7 = 8 * 128 = 1024 points
        final_ch = channels[-1]

        # Resize exactly to seq_len (512 -> 1000), then refine + project to 1 channel.
        self.resize = nn.Upsample(size=seq_len, mode="linear", align_corners=False)
        self.refine = nn.Sequential(
            nn.Conv1d(final_ch, final_ch, kernel_size=5, padding=2),
            nn.BatchNorm1d(final_ch),
            nn.LeakyReLU(0.2, inplace=True),
            nn.Conv1d(final_ch, 1, kernel_size=5, padding=2),
        )

        self.mse = nn.MSELoss()

    def forward(self, x):
        """A forward pass through the network."""
        # x: (B, input_dim)
        b = x.size(0)
        h = self.fc(x)  # (B, C*init_len)
        h = h.view(b, self.init_channels, self.init_len)  # (B, C, init_len)
        h = self.up_blocks(h)  # (B, C', ~512)
        h = self.resize(h)  # (B, C', seq_len)
        h = self.refine(h)  # (B, 1, seq_len)
        return h.squeeze(1)  # (B, seq_len)

class ResidualBlock(nn.Module):
    """The residual MLP block."""
    def __init__(self, dim):
        """
        Initialize the block.
        :param dim: The hidden dimension of the block's layers.
        """
        super().__init__()
        self.linear1 = nn.Linear(dim, dim)
        self.ln1 = nn.LayerNorm(dim)
        self.gelu = nn.GELU()

        self.linear2 = nn.Linear(dim, dim)
        self.ln2 = nn.LayerNorm(dim)



[docs]
    def forward(self, x):
        # Store the input for the skip connection.
        residual = x

        out = self.linear1(x)
        out = self.ln1(out)
        out = self.gelu(out)

        out = self.linear2(out)
        out = self.ln2(out)

        # Add the input back to the output before the final activation.
        out += residual
        return self.gelu(out)

class ResidualMLP(pl.LightningModule):
    """The residual MLP neural network module."""
    def __init__(self,
                 input_dim=18,
                 hidden_dim=512,
                 output_dim=1000,
                 num_blocks=3,
                 lr=1e-5,
                 weight_decay=1e-4,
                 signal_lambda=10):
        """
        Initialize the module.
        :param input_dim: The number of inputs.
        :param hidden_dim: The block's hidden dimension.
        :param output_dim: The number of outputs.
        :param num_blocks: The number of blocks.
        :param lr: The optimizer learning rate.
        :param weight_decay: The optimizer weight decay.
        :param signal_lambda: The loss function lambda parameter.
        """
        super().__init__()
        self.save_hyperparameters()

        # Project initial 30 features up to the hidden dimension.
        self.input_layer = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.LayerNorm(hidden_dim),
            nn.GELU()
        )

        # Stack the residual blocks.
        self.blocks = nn.ModuleList([ResidualBlock(hidden_dim) for _ in range(num_blocks)])

        # Final projection to 1000 outputs.
        self.output_layer = nn.Linear(hidden_dim, output_dim)
        self.mse = nn.MSELoss()



[docs]
    def forward(self, x):
        """A forward pass through the network."""
        x = self.input_layer(x)
        for block in self.blocks:
            x = block(x)
        return self.output_layer(x)
