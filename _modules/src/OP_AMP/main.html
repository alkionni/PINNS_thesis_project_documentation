def init_model(neural_network: str) -> nn.Module:
    """
    Initialize the model of the neural network.
    :param neural_network: The type of the neural network e.g. 'mlp'
    :return: The model.
    """
    if neural_network == 'mlp':
        model = MLP()
    elif neural_network == 'gru':
        model = SimpleGRU()
    elif neural_network == 'otm_gru':
        model = GRUOneToMany()
    elif neural_network == 'mto_gru':
        model = ManyToOneGru()
    else:
        model = None
        raise RuntimeError("Model was not initialized")
    return model

def train() -> None:
    """
    Train and save model.
    :return: None
    """
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(torch.cuda.get_device_name(0))

    dataset = load_dataset()
    model = init_model(NN).to(device)
    name = name_model(NN, dataset.batch, dataset.seq, BODE)
    lr = LR
    epochs = EPOCHS

    if RECORD:
        run = set_wandb(name)
        run.name = f"phase-mlp-v1"
    else:
        run = None
    if MODE == 'sweep':
        run = set_wandb(name)
        config = wandb.config
        lr = config.learning_rate
        # epochs = config.epochs
        run.name = f"lr={lr:.4f}"
    else:
        config = None

    # Train
    model.train()
    train_set = convert_input_to_tensor(get_train_set(dataset.norm_params, PERC), NN).to(device)

    if BODE == 'phase':
        train_data = dataset.norm_phase
    elif BODE == 'gain':
        train_data = dataset.norm_gain
    else:
        train_data = None
        print("Please specify whether you want phase or gain.")
    label = convert_output_to_tensor(get_train_set(train_data, PERC), NN).to(device)

    if VALIDATION:
        valid_set = convert_output_to_tensor(get_eval_set(dataset.norm_params, PERC, EVAL_PERC)[1], NN).to(device)
        val_label = convert_output_to_tensor(get_eval_set(train_data, PERC, EVAL_PERC)[1], NN).to(device)
        train_with_validation(model, train_set, label, valid_set, val_label,
                              lr, epochs, SAVE_DIR, name,
                              run, config)
    elif TRAIN_WITH_STEP:
        train_with_step_loop(model, train_set, label, lr, epochs, BATCH_SIZE, run, config)
        save_model(model, SAVE_DIR, name)
    else:
        train_loop(model, train_set, label, lr, epochs, run, config)
        save_model(model, SAVE_DIR, name)

    if run or config:
        wandb.finish()

def evaluate() -> None:
    """
    Load and evaluate model
    :return: None
    """
    # device = torch.device("cpu")
    print(torch.cuda.get_device_name(0))

    dataset = load_dataset()

    model = init_model(NN)  # .to(device)
    load_model(model, SAVE_DIR, SAVED_MODEL)

    if BODE == 'phase':
        test_data = dataset.phase
    elif BODE == 'gain':
        test_data = dataset.gain
    else:
        test_data = None
        print("Please specify whether you want phase or gain.")

    if VALIDATION:
        test = convert_input_to_tensor(get_eval_set(dataset.norm_params, PERC, EVAL_PERC)[0], NN)  # .to(device)
        label = get_eval_set(test_data, PERC, EVAL_PERC)[0]  # .to(device)
    else:
        test = convert_input_to_tensor(get_test_set(dataset.norm_params, PERC), NN)  # .to(device)
        label = get_test_set(test_data, PERC)

    # Test
    model.eval()

    prediction = model(test).detach().numpy()
    prediction = get_prediction(prediction, dataset, NN, BODE)

    # Results
    mse = get_mse(prediction, label)
    if 'gain' in SAVED_MODEL:
        bode_type = 'gain'
    elif 'phase' in SAVED_MODEL:
        bode_type = 'phase'
    else:
        bode_type = None
        print("Please specify whether you want phase or gain.")
    test = test.detach().numpy()
    plot_results(prediction, label, dataset, mse, bode_type, test)

def sweep() -> None:
    """
    Run a wandb sweep. Used for hyperparameter tuning.
    :return: None
    """
    sweep_config = {
        "method": "random",
        "metric": {"goal": "minimize", "name": "validation_loss"},
        "parameters": {
            "learning_rate": {"min": 0.0001, "max": 0.01},
        }
    }
    sweep_id = wandb.sweep(sweep_config, project='hyperparam-tuning-opamp')
    # Run the sweep
    wandb.agent(sweep_id, function=train, count=20)




if __name__ == "__main__":

        train()

        evaluate()
