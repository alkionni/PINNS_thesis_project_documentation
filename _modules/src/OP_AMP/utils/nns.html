class MLP(nn.Module):
    """The MLP neural network module."""
    def __init__(self):
        """Initialize the module."""
        super(MLP, self).__init__()
        self.network = nn.Sequential(
            nn.Linear(5, 100),
            nn.ReLU(),
            nn.Linear(100, 100),
            nn.ReLU(),
            nn.Linear(100, 100),
            nn.ReLU(),
            nn.Linear(100, 100)
        )

    def forward(self, x):
        """
        Forward pass: input shape (Batch, 5) -> output shape (Batch, 100)
        """
        out = self.network(x)
        return out

class SimpleGRU(nn.Module):
    def __init__(self, input_size=5, seq_len=1, hidden_size=128, num_layers=3, out_size=100):
        super().__init__()

        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.gru = nn.GRU(
            input_size,
            hidden_size,
            num_layers,
            batch_first=True
        )
        self.fc = nn.Linear(hidden_size*seq_len, out_size)

    def forward(self, x):
        h0 = torch.zeros(self.num_layers, x.size(0), self.hidden_size).to(x.device)
        out, _ = self.gru(x, h0)
        out = self.fc(out)
        return out

class GRUOneToMany(nn.Module):
    def __init__(self, input_size=5, seq_len=100, hidden_size=64, num_layers=2):
        super().__init__()
        self.num_layers = num_layers
        self.seq_len = seq_len
        self.init_fc = nn.Linear(input_size, hidden_size)
        self.gru = nn.GRU(1, hidden_size, num_layers, batch_first=True)
        self.out = nn.Linear(hidden_size, 1)



[docs]
    def forward(self, x, target=None, teacher_fr_ratio=0.5):
        batch_size = x.size(0)
        # Initiate hidden state
        h = self.init_fc(x).unsqueeze(0).repeat(self.num_layers, 1, 1).to(x.device)

        # Make dummy input from zeros
        input_step = torch.zeros(batch_size, 1, 1).to(x.device)

        # Begin autoregressive decoding
        outputs = []
        for t in range(self.seq_len):
            out, h = self.gru(input_step, h)
            y = self.out(out)
            outputs.append(y)

            # Decide next input with teacher forcing
            is_teacher = random.random() < teacher_fr_ratio
            if is_teacher and target is not None:
                # Ground truth
                input_step = target[:, t].unsqueeze(-1)
            else:
                # Inference (model's own prediction)
                input_step = y.detach()
        return torch.cat(outputs, dim=1)

class ManyToOneGru(nn.Module):
    def __init__(self, input_size=1, seq_len=5, hidden_size=128, num_layers=2, out_size=100):
        super().__init__()

        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.gru = nn.GRU(
            input_size,
            hidden_size,
            num_layers,
            batch_first=True
        )
        self.fc = nn.Linear(hidden_size, out_size)

    def forward(self, x):
        # x: (batch_size, seq_len, input_size)
        #    (n, 5, 1)
        # h0: (num_layers, batch_size, hidden_size)
        # out: (batch_size, seq_len, hidden_size) -> (batch_size, out_size)
        h0 = torch.zeros(self.num_layers, x.size(0), self.hidden_size).to('cpu')
        out, _ = self.gru(x, h0)
        # out = out.reshape(x.size(0), -1)
        out = out[:, -1, :]
        out = self.fc(out)
        return out
