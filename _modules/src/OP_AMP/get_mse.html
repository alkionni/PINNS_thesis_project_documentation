import os
import pandas as pd
from src.OP_AMP.utils.nns import MLP, SimpleGRU
from src.OP_AMP.utils.dataset import get_train_set, get_test_set, plot_results
from src.OP_AMP.utils.utils import convert_to_tensor
from src.OP_AMP.main import load_dataset, train_loop, get_prediction, get_mse

BODE_TYPE = "phase"
CSV_DIR = "mse"
CSV_FILENAME = f"{BODE_TYPE}_mse.csv"
LR = 0.0038
EPOCHS = 8000

def mse() -> None:
    """
    Calculate the average mse of a certain number of gru and mlp models for a
    given train size (that is a percentage of the dataset). Repeat for many
    different percentages. Save the average mses and sizes per nn in a csv.

    :return: None
    """
    dataset = load_dataset()
    gru_model = SimpleGRU()
    mlp_model = MLP()

    # Get mean mse every 5 trained models of a given train size and save in a csv
    columns = ["train set", "test set", "mlp mse"]  # , "gru mse"

    # Create save directory
    if not os.path.exists(f"{CSV_DIR}"):
        os.makedirs(f"./{CSV_DIR}")

    if not os.path.exists(f"{CSV_DIR}/{CSV_FILENAME}"):
        df_empty = pd.DataFrame(columns=columns)
        df_empty.to_csv(f"{CSV_DIR}/{CSV_FILENAME}", index=False)

    repeats = 1
    for perc in [0.72, 0.64, 0.56, 0.48, 0.4, 0.32, 0.24, 0.16, 0.08, 0.04]:
        mlp_total_mse = 0
        gru_total_mse = 0
        mlp_avg_mse = 0
        gru_avg_mse = 0
        counter = 0
        while counter < repeats:
            gru_model.train()
            mlp_model.train()

            mlp_train = convert_to_tensor(get_train_set(dataset.norm_params, perc), 'mlp')
            mlp_label = convert_to_tensor(get_train_set(dataset.norm_phase, perc), 'mlp')
            gru_train = convert_to_tensor(get_train_set(dataset.norm_params, perc), 'gru')
            gru_label = convert_to_tensor(get_train_set(dataset.norm_gain, perc), 'gru')

            print("MLP train loop:")
            train_loop(mlp_model, mlp_train, mlp_label, LR, EPOCHS)
            print("GRU train loop:")
            train_loop(gru_model, gru_train, gru_label, 0.002, 8000)

            mlp_model.eval()
            # gru_model.eval()
            mlp_test = convert_to_tensor(get_test_set(dataset.norm_params, 0.8),
                                         'mlp')
            mlp_label = get_test_set(dataset.phase, 0.8)
            gru_test = convert_to_tensor(get_test_set(dataset.norm_params, perc), 'gru')
            gru_label = get_test_set(dataset.gain, perc)

            mlp_prediction = mlp_model(mlp_test).detach().numpy()
            mlp_prediction = get_prediction(mlp_prediction, dataset, 'mlp', BODE_TYPE)
            mlp_mse = get_mse(mlp_prediction, mlp_label)
            gru_prediction = gru_model(gru_test).detach().numpy()
            gru_prediction = get_prediction(gru_prediction, dataset, 'gru', BODE_TYPE)
            gru_mse = get_mse(gru_prediction, gru_label)

            mlp_total_mse = mlp_total_mse + mlp_mse
            gru_total_mse = gru_total_mse + gru_mse

            # Plot
            plot_results(mlp_prediction, mlp_label, dataset, mlp_total_mse, BODE_TYPE)

            counter = counter + 1
            train_size = int(perc * dataset.batch)
            if counter == repeats:
                mlp_avg_mse = mlp_total_mse/repeats
                gru_avg_mse = gru_total_mse / repeats
                new_row = {
                    "train set": [train_size],
                    "test set": [dataset.batch - train_size],
                    "mlp mse": [mlp_avg_mse],
                    "gru mse": [gru_avg_mse]
                }
                df_row = pd.DataFrame(new_row)
                df_row.to_csv(f"{CSV_DIR}/{CSV_FILENAME}", mode='a', index=False, header=False)
                print(df_row)




if __name__ == "__main__":
    mse()
