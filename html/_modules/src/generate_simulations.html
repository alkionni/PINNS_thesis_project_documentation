def lhs(n_params, n_samples, ta_oria):
    """
    Latin Hypercube Sampling for LDO.
    :param n_params: The number of input params.
    :param n_samples: The number of samples to generate.
    :param ta_oria: The range of every input param.
    :return:
    """
    sampler = qmc.LatinHypercube(d=n_params, seed=0)
    sample = sampler.random(n=n_samples)

    low_lim = list()
    high_lim = list()
    for param in ta_oria:
        low_lim.append(ta_oria[param][0])
        high_lim.append(ta_oria[param][1])

    return np.array(qmc.scale(sample, low_lim, high_lim), dtype=np.float32)

def opamp_sim_light(file_name: str):
    """
    Generate simulations for every input combination. This is a "light version"
    meaning it will not store information that is not necessary.
    :param file_name: the name of the pickle file to write the data
    :return: None
    """
    config_path = "./OP_AMP/config/65nm/"
    to_save_path = '../simulation_data'

    # Ensure the save path exists and is clean
    if os.path.exists(to_save_path):
        shutil.rmtree(to_save_path)
    os.makedirs(to_save_path)

    # Load configs
    # Note: Ensure these JSON files exist relative to where you run the test
    circuit_config, _ = si.load_configs(
        f"{config_path}circuit.json",
        f"{config_path}training.json"
    )

    """
        Get simulation data for various inputs.
    """
    print("\n=== Preparing to run simulation ===")
    scs_related_files_name = 'generate_data'

    # Create SCS file
    scs_files_path = si.create_scs_file(
        to_save_path,
        scs_related_files_name,
        circuit_config['technology']['scs_template']
    )

    # Generate design parameters
    Cc_in = ['1.0p', '5.0p', '10.0p', '15.0p', '20.0p']
    Wbp_in = ['1.0u', '5.0u', '10.0u', '15.0u', '20.0u']
    Wdiff_in = ['0.3u', '1.0u', '2.0u', '3.3u']
    Wnb_in = ['1.0u', '5.0u', '10.0u', '15.0u', '20.0u']
    Wpo_in = ['10.0u', '20.0u', '30.0u', '40.0u']

    sims = (len(Cc_in) * len(Wbp_in) * len(Wdiff_in) * len(Wnb_in) * len(Wpo_in))
    dict_in = {}
    dict_out = {}
    print_every = 100
    data_from_simulation = []
    for i in range(len(Cc_in)):
        for j in range(len(Wbp_in)):
            for k in range(len(Wdiff_in)):
                for m in range(len(Wnb_in)):
                    for n in range(len(Wpo_in)):
                        # Arrange parameters in a dict
                        dict_in['Cc'] = Cc_in[i]
                        dict_in['Wbp'] = Wbp_in[j]
                        dict_in['Wdiff'] = Wdiff_in[k]
                        dict_in['Wnb'] = Wnb_in[m]
                        dict_in['Wpo'] = Wpo_in[n]
                        params_to_write = dict_in.copy()

                        # Write parameters
                        parameters_scs_path = f"{scs_files_path}/{scs_related_files_name}_parameters.scs"

                        # params_to_write =
                        # [{'Cc': '12.0p', 'Wbp': '11.0u', 'Wdiff': '10.0u', 'Wnb': '1.0u', 'Wpo': '20.0u'}]

                        # Get data from simulations
                        si.write_params(parameters_scs_path, params_to_write)

                        # Run Spectre
                        scs_to_execute = f"{scs_files_path}/{scs_related_files_name}.scs"
                        results_path = si.run_spectre(scs_to_execute, to_save_path)

                        # Calculate Metrics
                        metrics_obj = OpAmp(
                            circuit_config,
                            scs_files_path,
                            experiment_name=scs_related_files_name
                        )
                        metrics_obj.get_simulation_data(results_path)
                        metrics_obj.update_oppoint_results()

                        frequency = metrics_obj.frequency
                        frequency = np.array(frequency, dtype=np.float64)
                        gain = metrics_obj.gain().magnitude(db=True)
                        phase = metrics_obj.gain().phase() + 180

                        if not data_from_simulation:
                            dict_out['frequency'] = frequency
                        dict_out['input'] = params_to_write
                        dict_out['gain'] = gain
                        dict_out['phase'] = phase
                        data_from_simulation.append(dict_out.copy())
                        dict_out.clear()

                        if len(data_from_simulation) % print_every == 0:
                            print(f"Simulation {len(data_from_simulation)}/{sims}")

                        dict_in.clear()

    # Save data to a pickle file
    models_dir = "saved_simulation_data"
    timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")

    if not os.path.exists(models_dir):
        os.makedirs(models_dir)
    with open(f"{models_dir}/{file_name}_{timestamp}.pkl", "wb") as f:
        pickle.dump(data_from_simulation, f)

    # Cleanup stray files in current directory
    for file in os.listdir():
        if file.endswith(".dc") and os.path.isfile(file):
            os.remove(file)
    for folder in os.listdir():
        if folder.endswith(".ahdlSimDB") and os.path.isdir(folder):
            shutil.rmtree(folder)

def ldo_sim_light(file_name: str, samples: int, seq_points: int) -> None:
    """
    Generate simulations for the LDO circuit, without time included.
    :param file_name: the name of the generated pickle file
    in which the simulation results are saved into
    :param samples: the number of simulations to create
    :param seq_points: the number of points per graph
    :return: None
    """

    # The netlist is the absolute path to the netlist.scs file
    netlist = "./LDO/config/input.scs"
    to_save_path = 'simulation_data'

    # Ensure the save path exists and is clean
    if os.path.exists(to_save_path):
        shutil.rmtree(to_save_path)
    os.makedirs(to_save_path)

    """
    Simulate and save results.
    """
    print("\n=== Preparing to run simulation ===")
    scs_related_files_name = 'generate_data'

    # Create SCS file
    scs_files_path = si.create_scs_file(
        to_save_path,
        scs_related_files_name,
        netlist
    )
    # scs_files_path = to_save_path

    # File path to parameters
    parameters_scs_path = f"{scs_files_path}/{scs_related_files_name}_parameters.scs"

    # Generate sample with latin hypercube space
    num_of_params_to_sample = 18
    params = lhs(num_of_params_to_sample, samples, mikrotera_oria)

    # Prepare the file to save
    models_dir = "saved_simulation_data"

    # Initialize logging
    logging.basicConfig(
        filename='file.log',
        level=logging.INFO,
        format='%(asctime)s - %(levelname)s - %(message)s'
    )
    logging.info("\n")

    if not os.path.exists(models_dir):
        os.makedirs(models_dir)

    with h5py.File(f"{models_dir}/{file_name}.h5", "a") as f:
        # Create datasets (chunks=True enables faster writing/compression)
        if 'input' not in f:
            d_0 = f.create_dataset("input", (samples, num_of_params_to_sample), dtype=np.float32, chunks=True)
        if 'ldr' not in f:
            d_2 = f.create_dataset("ldr", (samples, seq_points), dtype=np.float32, chunks=True)
        if 'lnr' not in f:
            d_3 = f.create_dataset("lnr", (samples, seq_points), dtype=np.float32, chunks=True)

        print_every = 100
        for i in range(samples):
            # Params that are constant will take values according to the original design.
            params_to_write = {
                'BF_l1': 500e-09, 'BF_l2': 350e-09, 'BF_l3': 1.2e-06, 'BF_l4': 2e-06,
                'BN_l1': 1.3e-06, 'BN_l2': 12.5e-06, 'BN_l3': 12.5e-06, 'BN_l4': 1.3e-06,
                'BR_pl': 30e-06, 'BR_nl':  30e-06,
                'OT_l1': 800e-09, 'OT_l2': 2e-06, 'OT_l3': 4e-06,
                'TC_l1': 350e-09, 'TC_l2': 300e-09, 'TC_l3': 350e-09, 'TC_l4': 300e-09, 'TC_l5': 350e-09,
                'TC_l6': 300e-09, 'TC_l7': 350e-09, 'TC_l8': 300e-09, 'TC_l9': 350e-09, 'TC_l10': 300e-09,
                'VDD': 3.3, 'CL': 50e-12, 'RL': 60.0, 'Vref': 0.9, 'Delay': 20e-06, 'Period': 1.0,
                'Pulse_Width': 80e-06, 'Risetime': 1e-06, 'Falltime': 1e-06, 'VDD_min': 2.1, 'Fin': 1e03,
                'Cc1': params[i][0], 'Cc2': params[i][1], 'PASS_FingW': params[i][2], 'PASS_l': params[i][3],
                'BF_w1': params[i][4], 'BF_w2': params[i][5], 'BF_w3': params[i][6], 'BF_w4': params[i][7],
                'BN_w1': params[i][8], 'BN_w2': params[i][9], 'BN_w3': params[i][10], 'BN_w4': params[i][11],
                'BN_w5': params[i][12],
                'BR_pw': 420e-09, 'BR_nw': 420e-09,
                'FN_R': params[i][13],
                'OT_w1': params[i][14], 'OT_w2': params[i][15], 'OT_w3': params[i][16],
                'R_B2': params[i][17],
                'TC_w1': 3.36e-06, 'TC_w2': 420e-09, 'TC_w3': 84e-06, 'TC_w4': 840e-09, 'TC_w5': 840e-09,
                'TC_w6': 10.08e-06, 'TC_w7': 8.4e-06, 'TC_w8': 600e-09, 'TC_w9': 840e-09, 'TC_w10': 420e-09
            }
            si.write_params(parameters_scs_path, params_to_write)

            # Run Spectre
            scs_to_execute = f"{scs_files_path}/{scs_related_files_name}.scs"
            results_path = si.run_spectre(scs_to_execute, to_save_path)

            # Read the outputs
            exodoi = ipol(results_path)

            time = np.array(exodoi['tran']['time'], dtype=np.float32)
            ldr = np.array(exodoi['tran']['Vout\#28t\#29'], dtype=np.float32)
            lnr = np.array(exodoi['tran']['Vout_LNR\#28t\#29'], dtype=np.float32)

            # # -------- pchip interpolate -------------
            t_uniform = np.linspace(0, 0.00015, 15000)
            ldr_uniform = pchip_interpolate(time, ldr, t_uniform)
            lnr_uniform = pchip_interpolate(time, lnr, t_uniform)

            indices_lnr = np.linspace(0, len(ldr_uniform) - 1, seq_points).astype(int)
            indices_ldr = np.linspace(0, len(lnr_uniform) - 1, seq_points).astype(int)
            ldr_uniform = ldr_uniform[indices_ldr]
            lnr_uniform = lnr_uniform[indices_lnr]

            f['input'][i] = params[i]
            f['ldr'][i] = ldr_uniform
            f['lnr'][i] = lnr_uniform

            print(f"{len(time)} index {i}")
            logging.info(f'{len(time)} index {i}')

            if len(time) > 5000:
                print(f"PROBLEM! Points: {len(time)}. Index: {i}")
                logging.info(f"PROBLEM! Points: {len(time)}. Index: {i}")

            if i % print_every == 0:
                print(f"Simulation {i}/{len(params)}")

    # Cleanup stray files in current directory
    for file in os.listdir():
        if file.endswith(".dc") and os.path.isfile(file):
            os.remove(file)
    for folder in os.listdir():
        if folder.endswith(".ahdlSimDB") and os.path.isdir(folder):
            shutil.rmtree(folder)
