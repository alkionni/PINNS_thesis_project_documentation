def initialize_model(path: str):
    """Initialize the model."""
    device = torch.device("cpu")
    model = ResidualMLP.load_from_checkpoint(path)
    model.eval()
    return model.to(device)

def ldr_transient_model(model, eisodoi) -> np.array:
    """
    Predicts the load transient response based on the input.
    :param model: The LDR model.
    :param eisodoi: The input.
    :return: The prediction.
    """
    # Transform input.
    feature = np.zeros(eisodoi.shape)
    for i, param in enumerate(config.ORIA):
        feature[i] = \
            (2 * (eisodoi[i] - config.ORIA[param][0]) /
             (config.ORIA[param][1] -
              config.ORIA[param][0]) - 1)
    feature = torch.tensor(feature, dtype=torch.float32)

    # Make prediction.
    predictions = model(feature)

    # Inverse transform output.
    predictions = predictions.detach().cpu().numpy()
    predictions = (predictions + 1) / 2 * (config.LDR_TARGET[1] - config.LDR_TARGET[0]) + config.LDR_TARGET[0]

    return predictions

def ldr_lnr_inference(eisodoi):
    """
    Predicts the LDO transient response based on the input.
    :param eisodoi: The input.
    :return: The prediction (and also time).
    """
    # Initialize models.
    ldr_model = initialize_model(config.LDR_PATH)
    lnr_model = initialize_model(config.LNR_PATH)

    # Make predictions.
    pred_ldr = ldr_transient_model(ldr_model, eisodoi)
    pred_lnr = lnr_transient_model(lnr_model, eisodoi)

    timestep = np.linspace(0, 0.00016, 1000)
    return timestep, pred_ldr, pred_lnr
