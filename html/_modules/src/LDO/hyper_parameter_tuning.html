def objective(trial, logger, max_epochs) -> float:
    """
    Create the hyperparameter sweep's objective.
    :param trial: The current trial.
    :param logger: Whether to use logger or not.
    :param max_epochs: The number of epochs to train for.
    :return: The validation loss of the trial.
    """
    # Set the hyperparameters to optimize.
    lr = trial.suggest_float('lr', 1e-5, 1e-1, log=True)
    batch_size = trial.suggest_categorical('batch_size', [512, 1024, 2048])
    weight_decay = trial.suggest_float('weight_decay', 1e-5, 1e-1, log=True)  # lower best for shallow mlps
    signal_lambda = trial.suggest_float('signal_lambda', 1, 20)
    # hidden_dim = trial.suggest_categorical('hidden_dim', [128, 256, 512])  # res-mlp
    # hidden_dim = trial.suggest_categorical('hidden_dim', [64, 128, 256])  # cnn
    # num_blocks = trial.suggest_categorical('num_blocks', [2, 3, 5])

    # Create the model with trial hyperparameters.
    model = CNNRegressor(
        lr=lr,
        weight_decay=weight_decay,
        signal_lambda=signal_lambda,
        # hidden_dim=hidden_dim,
        # num_blocks=num_blocks
    )

    # Optuna pruning callback.
    pruning_callback = PyTorchLightningPruningCallback(trial, monitor='val_loss')

    # Wandb is used as the Logger.
    if logger:
        logger = WandbLogger(
            project="project-name",
            name=f"trial_{trial.number}",
            group="optuna-tuning",
            log_model=False
        )

        logger.log_hyperparams(trial.params)

    # Initialize trainer.
    trainer = pl.Trainer(
        logger=logger,
        accelerator='gpu',
        devices=1,
        max_epochs=max_epochs,
        precision=32,
        callbacks=[
            MyPrintingCallback(),
            DisableValProgressBar(),
            pruning_callback
        ],
        enable_checkpointing=False
    )

    # Initialize data module.
    dm = DataModule(
        num_workers=0,
        path=config.DATASET_PATH,
        dataset_size=config.DATASET_SIZE,
        oria=config.ORIA,
        batch_size=batch_size,
        label=config.LABEL
    )

    # Train the model.
    trainer.fit(model, dm)

    # Finish trial to keep each one as a separate run.
    import wandb
    wandb.finish()

    # Final validation loss.
    return trainer.callback_metrics['val_loss'].item()

def run_optimization(n_trials=50, logger=True, max_epochs=config.TUNING_EPOCHS):
    """
    Configure the hyperparameter sweep.
    :param n_trials: Number of trials.
    :param logger: Whether to use logger or not.
    :param max_epochs: The number of epochs to train for.
    :return: The study.
    """
    # Create a pruner for bad trials.
    pruner = optuna.pruners.MedianPruner(n_startup_trials=5, n_warmup_steps=10)

    # Initialize and run a study.
    # Lambda wrap is because I wanted to pass "logger" argument to objective().
    func = lambda trial: objective(trial, logger, max_epochs)

    study = optuna.create_study(direction='minimize', pruner=pruner)
    study.optimize(func, n_trials=n_trials)

    # Print results.
    print("Best trial:")
    trial = study.best_trial
    print(f"    Value: {trial.value}")
    print(f"    Params: ")
    for key, value in trial.params.items():
        print(f"    {key}: {value}")

    return study




if __name__ == "__main__":
    # Run hyperparameter sweep.
    study_res = run_optimization(n_trials=config.TUNING_TRIALS)
