def get_mse(prediction: np.array, label: np.array) -> float:
    """
    Get the mean square error of prediction and actual curve.
    """
    return np.mean((prediction - label) ** 2)

def convert_to_tensor(array: np.array, neural_network: str) -> torch.tensor:
    """
    Convert to tensor based on the needs of the neural network.
    :param array: The array to convert.
    :param neural_network: The type of neural network e.g. 'mlp'.
    :return: The array converted to tensor.
    """
    if neural_network == 'mlp':
        return torch.tensor(array, dtype=torch.float32)
    elif neural_network == 'gru':
        return torch.tensor(array, dtype=torch.float32).unsqueeze(1)
    else:
        return None

def convert_input_to_tensor(array: np.array, neural_network: str) -> torch.tensor:
    """
    Convert to tensor based on the needs of the neural network.
    :param array: The input to the neural network.
    :param neural_network: The type of neural network e.g. 'mlp'.
    :return: The array converted to tensor.
    """
    if neural_network == 'mlp' or neural_network == 'otm_gru':
        return torch.tensor(array, dtype=torch.float32)
    elif neural_network == 'gru':
        return torch.tensor(array, dtype=torch.float32).unsqueeze(1)
    elif neural_network == 'mto_gru':
        return torch.tensor(array, dtype=torch.float32).unsqueeze(-1)
    else:
        return None

def name_model(neural_network: str, batch: int, seq: int, bode: str) -> str:
    """
    Name the model to be trained, so that it can be saved with that name.
    :param neural_network: The type of the neural network e.g. 'mlp'.
    :param batch: The size of the dataset.
    :param seq: The sequence length of the curves.
    :param bode: The type of curve e.g. phase
    :return: The name of the model to be saved.
    """
    timestamp = datetime.now().strftime("%d-%m-%Y_%H-%M-%S")
    return f"{bode}_{neural_network}_{seq}_{batch}_{timestamp}.pth"

def save_model(model: nn.Module, save_dir: str, name: str) -> None:
    """
    Save the model.
    :param model: The model.
    :param save_dir: The directory to save the model in.
    :param name: The name of the file to be saved.
    :return: None
    """
    if not os.path.exists(save_dir):
        os.makedirs(save_dir)

    torch.save(model.state_dict(), f"{save_dir}/{name}")

def train_loop(model: nn.Module, train: torch.tensor, label: torch.tensor, lr: float,
               epochs: int, run=None, config=None) -> None:
    """
    The loop used to train the neural network.
    :param model: The initialized model to train.
    :param train: The train set of the dataset to be used as input.
    :param label: The labels to be used for backpropagation.
    :param lr: The learning rate.
    :param epochs: The number of epochs.
    :param run: The run of wandb used for logging the training loss.
    :param config: The config of wandb used for logging the training loss for sweeps.
    :return: None
    """
    optimizer = torch.optim.Adam(model.parameters(), lr=lr)
    criterion = nn.MSELoss()
    start_time = datetime.now()
    print("Begin Training...")
    for epoch in range(epochs):
        optimizer.zero_grad()

        prediction = model(train)
        loss = criterion(prediction, label)
        if run:
            run.log({"training_loss": loss})
        if config:
            wandb.log({"training_loss": loss})

        loss.backward()
        optimizer.step(None)

        if (epoch + 1) % 100 == 0:
            print(f"Epoch {epoch + 1}/{epochs}, "
                  f"Data Loss = {loss:.6f}")
    end_time = datetime.now()
    print(f"Elapsed time: {end_time-start_time}")

def train_with_validation(model: nn.Module, train: torch.tensor, label: torch.tensor,
                          valid: torch.tensor, val_label: torch.tensor,
                          lr: float, epochs: int, save_dir: str, name: str,
                          run=None, config=None) -> None:
    """
    The loop used to train the neural network with validation.
    :param model: The initialized model to train.
    :param train: The train set of the dataset to be used as input.
    :param label: The labels to be used for backpropagation.
    :param valid: The validation input.
    :param val_label: The validation label.
    :param lr: The learning rate.
    :param epochs: The number of epochs.
    :param save_dir: The dir to save the model.
    :param name: The name of the file to save the model.
    :param run: The run of wandb used for logging the training loss.
    :param config: The config of wandb used for logging the training loss for sweeps.
    :return: None
    """
    optimizer = torch.optim.Adam(model.parameters(), lr=lr)
    criterion = nn.MSELoss()
    min_val_loss = np.inf
    for epoch in range(epochs):
        model.train()
        optimizer.zero_grad()

        prediction = model(train)
        loss = criterion(prediction, label)

        loss.backward()
        optimizer.step(None)

        # --------- Validation --------------
        model.eval()
        target = model(valid)
        valid_loss = criterion(target, val_label)

        if run:
            run.log({"training_loss": loss,
                     "validation_loss": valid_loss})
        if config:
            wandb.log({"training_loss": loss})
            wandb.log({"validation_loss": valid_loss})

        if (epoch + 1) % 100 == 0:
            print(f"Epoch {epoch + 1}/{epochs}, "
                  f"Data Loss = {loss:.6f}, Valid_loss = {valid_loss: .6}")
            if min_val_loss > valid_loss:
                print(f"Validation Loss Decreased "
                      f"{min_val_loss:.6f} --> {valid_loss:.6f}. \t Saving model...")
                save_model(model, save_dir, name)
        save_model(model, save_dir, f"last_{name}")

def train_with_step_loop(model: nn.Module, train: torch.tensor, label: torch.tensor, lr: float,
                         epochs: int, batch_size: int, neu_n: str, run=None, config=None) -> None:
    """
    The loop used to train the neural network with smaller batch and steps.
    :param model: The initialized model to train.
    :param train: The train set of the dataset to be used as input.
    :param label: The labels to be used for backpropagation.
    :param lr: The learning rate.
    :param epochs: The number of epochs.
    :param batch_size: The batch size.
    :param neu_n: The type of neural network.
    :param run: The run of wandb used for logging the training loss.
    :param config: The config of wandb used for logging the training loss for sweeps.
    :return: None
    """
    optimizer = torch.optim.Adam(model.parameters(), lr=lr)
    criterion = nn.MSELoss()
    start_time = datetime.now()
    total_steps = int(len(train)/batch_size)
    print("Begin Training...")
    for epoch in range(epochs):
        for step in range(total_steps):
            optimizer.zero_grad()

            if neu_n == 'otm_gru':
                prediction = model(train[:batch_size], label[:batch_size],
                                   teacher_fr_ratio=0.8)
            else:
                prediction = model(train[:batch_size])
            loss = criterion(prediction, label[:batch_size])
            if run:
                run.log({"training_loss": loss})
            if config:
                wandb.log({"training_loss": loss})

            loss.backward()
            optimizer.step(None)

            if (epoch + 1) % 10 == 0:
                print(f"Epoch {epoch + 1}/{epochs}, "
                      f"Data Loss = {loss:.6f}")
    end_time = datetime.now()
    print(f"Elapsed time: {end_time-start_time}")

def set_wandb(name: str) -> wandb.Run:
    """
    Initialize a wandb run to log results.
    :param name: The name of the model's file.
    :return: The run.
    """

        entity="sisisoucou-auth",
        project="op-amp-ml",
        job_type="training",
        config={
            "dataset_name": "data_2000_2026-02-09_22-56-51.pkl",
            "model_name": name
        }
    )
    return run
