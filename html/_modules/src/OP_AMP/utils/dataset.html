def get_dataset_from_file(path: str) -> list[dict]:
    """
    Load the raw dataset from the pickle file.
    :return: The raw dataset.
    """
    with open(path, 'rb') as f:
        dataset = pickle.load(f)

    if not dataset:
        print("The dataset is empty")

    return dataset

def get_params(dataset: list[dict]) -> np.array:
    """
    Get the input parameters that were used to get the simulations
    :param dataset: The raw dataset.
    :return: The parameters.
    """
    params = []
    for i in range(len(dataset)):
        for key in dataset[i]['input']:
            s = dataset[i]['input'][key]
            for char, replacement in UNITS.items():
                if char in s:
                    s = s.replace(char, replacement)
            params.append(np.float64(s))
    return np.array(params).reshape(len(dataset), len(dataset[0]['input']))

class Dataset:
    """
    A handler for the dataset used for training and evaluation.
    """
    def __init__(self):
        self.raw_dataset = get_dataset_from_file(PATH)
        self.batch = get_dataset_size(self.raw_dataset)
        self.seq = get_sequence_size(self.raw_dataset)

        self.frequency = get_frequency(self.raw_dataset)
        self.params = get_params(self.raw_dataset)
        self.gain = get_gain(self.raw_dataset)
        self.phase = get_phase(self.raw_dataset)

        self.norm_params = None
        self.norm_gain = None
        self.norm_phase = None

        self.gain_range = get_range(self.gain)
        self.phase_range = get_range(self.phase)

    def reduce_sequence_points(self, num: int) -> None:
        """Reduce sequence points of gain and phase."""
        indices = np.linspace(0, len(self.raw_dataset[0]['frequency']) - 1, num).astype(int)
        self.frequency = self.frequency[indices]
        self.gain = self.gain[:, indices]
        self.phase = self.phase[:, indices]
        self.seq = num

def normalize(array: np.array, array_range: list[np.array]) -> np.array:
    """
    Normalize the array with min-max.
    :param array: An array e.g. phase.
    :param array_range: The range of the array (min and max values).
    :return: The normalized array.
    """
    return 2 * (array - array_range[0]) / (array_range[1] - array_range[0]) - 1

def denormalize(array: np.array, array_range: list[np.array]) -> np.array:
    """
    Denormalize the array with min-max.
    :param array: An array e.g. phase.
    :param array_range: The range of the array (min and max values).
    :return: The array presented with its real values.
    """
    return (array + 1) / 2 * (array_range[1] - array_range[0]) + array_range[0]

def get_mean_across_batch(array: np.array) -> np.array:
    """

    :param array: The range of the array (min and max values).
    :return: The mean curve (e.g. phase)

def shuffle_indices(batch_size: int) -> np.array:
    """
    Shuffle the indices (per batch) of an array.
    :param batch_size: The size of the array.
    :return: The shuffled indices.
    """
    np.random.seed(0)
    return np.random.permutation(batch_size)

def get_train_set(array: np.array, percentage: float) -> np.array:
    """
    Get the train set, which is a percentage of the dataset, to be used for the training.
    :param array: The array to train e.g. phase.
    :param percentage: The percentage of the dataset to keep as the train set.
    :return: The train array.
    """
    indices = shuffle_indices(len(array))
    train_size = int(percentage * len(indices))
    train_idx = indices[:train_size]
    return array[train_idx]

def get_test_set(array: np.array, percentage: float) -> np.array:
    """
    Get the test set, which is a percentage of the dataset, to be used for the evaluation.
    :param array: The array to test e.g. phase.
    :param percentage: The percentage of the dataset to keep as the test set.
    :return: The test array.
    """
    indices = shuffle_indices(len(array))
    train_size = int(percentage * len(indices))
    test_idx = indices[train_size:]
    return array[test_idx]

def get_eval_set(array: np.array, train_perc: float, eval_perc: float) -> np.array:
    """
    Get the validation set, which is a percentage of the dataset, to be used during training.
    :param array: The array to test e.g. phase.
    :param train_perc: The percentage of the dataset to divide for training (the rest is evaluation set).
    :param eval_perc: The percentage of the evaluation set to divide for validation.
    :return: The validation array.
    """
    indices = shuffle_indices(len(array))
    train_size = int(train_perc * len(indices))
    eval_idx = indices[train_size:]

    val_size = int(eval_perc * len(eval_idx))
    test_idx = eval_idx[:val_size]
    val_idx = eval_idx[val_size:]
    return array[test_idx], array[val_idx]

def get_prediction(raw_prediction: np.array, dataset: Dataset, neural_network: str, bode: str) -> np.array:
    """
    Get the prediction of the neural network and denormalize the output.
    :param raw_prediction: The prediction in normalized form.
    :param dataset: The dataset.
    :param neural_network: The type of the neural network e.g. 'gru'.
    :param bode: The type of the curve e.g. phase.
    :return: The denormalized prediction of the curve.
    """
    if bode == 'phase':
        array_range = dataset.phase_range
    elif bode == 'gain':
        array_range = dataset.gain_range
        bode = 'magnitude'
    else:
        array_range = None
        print("Please specify whether you want phase or gain.")
    prediction = denormalize(raw_prediction, array_range)

    if neural_network == "mlp" or neural_network == 'mto_gru':

        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
        fig.suptitle(f'Distribution of {bode} prediction before and after de-normalization', fontsize=14)
        bins = np.linspace(raw_prediction.min(), raw_prediction.max(), 100)
        ax1.set_title(f"Before")
        ax1.set_xlabel("LDR (per unit)")
        ax1.hist(raw_prediction.flatten(), bins, edgecolor='black', alpha=0.7)
        ax1.set_xscale('linear')

        bins = np.linspace(prediction.min(), prediction.max(), 100)
        ax2.set_title(f"After")
        ax2.set_xlabel("LDR (Volt)")
        ax2.hist(prediction.flatten(), bins, edgecolor='black', alpha=0.7)
        ax2.set_xscale('linear')
        plt.show()

        return prediction
    elif neural_network == "gru":
        return np.squeeze(prediction, 1)
    elif neural_network == "otm_gru":
        return np.squeeze(prediction, -1)
    else:
        print("You need a valid model in order to get prediction")
        return None

def plot_results(prediction: np.array, label: np.array, dataset: Dataset,
                 mse: float, bode_type: str, test=None) -> None:
    """
    Plot the prediction of the model along with the actual curve, the dataset mean and display the mse.
    :param prediction: The predicted curve.
    :param label: The actual curve.
    :param dataset: The dataset.
    :param mse: The mean square error of predicted and actual curve.
    :param bode_type: The type of the curve e.g. phase
    :param test: The test input.
    :return: None
    """
    if bode_type == 'gain':
        mean = get_mean_across_batch(dataset.gain)
        bode_type = 'magnitude'
    elif bode_type == 'phase':
        mean = get_mean_across_batch(dataset.phase)
    else:
        mean = None

    # Denormalize inputs
    params = np.empty((len(test), len(test[0])))
    params[:, 0] = denormalize(test[:, 0], get_range(dataset.params[:, 0]))
    params[:, 1] = denormalize(test[:, 1], get_range(dataset.params[:, 1]))
    params[:, 2] = denormalize(test[:, 2], get_range(dataset.params[:, 2]))
    params[:, 3] = denormalize(test[:, 3], get_range(dataset.params[:, 3]))
    params[:, 4] = denormalize(test[:, 4], get_range(dataset.params[:, 4]))

    t_mse = np.mean((prediction - label) ** 2)
    t_rmse = np.sqrt(np.mean((prediction - label) ** 2))
    t_mae = mean_absolute_error(label, prediction)
    t_mape = mean_absolute_percentage_error(label, prediction)

    tot_plots = len(prediction)
    for i in range(1):
        mse = np.mean((prediction[i] - label[i]) ** 2)
        rmse = np.sqrt(np.mean((prediction[i] - label[i]) ** 2))
        mae = mean_absolute_error(label[i], prediction[i])
        mape = mean_absolute_percentage_error(label[i], prediction[i])
        plt.figure(figsize=(12, 5))
        # print(f'inputs: {params[i]}')
        plt.suptitle(f'Total mse: {t_mse}, mae: {t_mae}, mape: {t_mape}, rmse: {t_rmse}', fontsize=12)
        plt.title(f'Plot {i + 1}/{tot_plots}, mse: {mse}, mae: {mae}, mape: {mape}, rmse: {rmse}', fontsize=10)
        plt.plot(dataset.frequency, label[i], color='blue')
        plt.plot(dataset.frequency, prediction[i], color='red')
        plt.plot(dataset.frequency, mean, color='green', linestyle='--')
        plt.xlabel('frequency')
        plt.ylabel(f'{bode_type}')
        plt.xscale('log')
        plt.legend(['True', 'Predicted', 'Dataset mean'])
        plt.grid(True, which='both', ls='-', alpha=0.5)

        plt.tight_layout()
        plt.show()
