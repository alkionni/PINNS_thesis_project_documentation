import pandas as pd
import matplotlib.pyplot as plt
import os

GAIN_FILE = 'mse/gain_mse.csv'
PHASE_FILE = 'mse/phase_mse.csv'

def compare_mse() -> None:
    """
    Compare the performance of the mlp and gru neural network based on mse, for decreasing
    training size set.
    """
    df1 = pd.read_csv(GAIN_FILE)
    df2 = pd.read_csv(PHASE_FILE)
    print(df1)
    print(df2)

    x = df1['train set']

    plt.figure(figsize=(12, 5))
    plt.title("MSE on decreasing train set size")
    plt.xlabel("train set size")
    plt.ylabel("MSE")
    plt.scatter(x, df1['mlp mse'],
                color='blue', label='gain')
    plt.scatter(x, df2['mlp mse'],
                color='red', label='phase')
    plt.legend()
    plt.grid(True, which='both', ls='-', alpha=0.5)
    plt.tight_layout()
    plt.gca().invert_xaxis()
    plt.show()

def plot_mse():
    """Check for non-existing directories."""
    if os.path.exists(GAIN_FILE) and os.path.exists(PHASE_FILE):
        compare_mse()
    else:
        print("Missing gain and phase mse csv files. If mse directory "
              "doesn't exist you first need to run get_mse.py to generate "
              "both the directory and the csv files. If the directory exists "
              "somewhere else, you need to update the paths in the top of "
              "the plot_mse.py file.")




if __name__ == "__main__":
    plot_mse()
